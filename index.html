// 1. 讀取全校學生與學期資料 API function doGet(e) { e = e || { parameter: {} }; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; var data = sheet.getDataRange().getValues(); var result = {}; // 從第 2 列開始讀取 (第 1 列為表頭) for (var i = 1; i < data.length; i++) { var studentId = String(data[i][0]).trim(); var studentName = data[i][1]; var grade = data[i][2] || 1; // 讀取年級 (1:高一, 2:高二, 3:高三) var semestersJson = data[i][3]; if (studentId) { try { result[studentId] = { id: studentId, name: studentName, grade: parseInt(grade), semesters: semestersJson ? JSON.parse(semestersJson) : {} }; } catch (err) { result[studentId] = { id: studentId, name: studentName, grade: parseInt(grade), semesters: {} }; } } } return ContentService.createTextOutput(JSON.stringify({ status: "success", data: result })).setMimeType(ContentService.MimeType.JSON); } // 2. 新增學生 / 儲存更新 IEP 資料 API function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; var payload = JSON.parse(e.postData.contents); var data = sheet.getDataRange().getValues(); // 初始化雲端試算表表頭 (欄位包含：學號, 姓名, 年級, 所有學期JSON, 最後更新時間) if (data.length === 0 || !data[0][0]) { sheet.appendRow(["student_id", "student_name", "grade", "semesters_json", "updated_at"]); data = sheet.getDataRange().getValues(); } var studentId = String(payload.student_id).trim(); var studentName = payload.student_name || ""; var grade = payload.grade || 1; var semestersData = payload.semesters || {}; var rowIndex = -1; // 搜尋是否已有該學號學生 for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === studentId) { rowIndex = i + 1; // 試算表行號從 1 開始 break; } } var now = new Date().toLocaleString("zh-TW", { timeZone: "Asia/Taipei" }); if (rowIndex > 0) { // 覆蓋更新原有學生列 sheet.getRange(rowIndex, 2).setValue(studentName); sheet.getRange(rowIndex, 3).setValue(grade); sheet.getRange(rowIndex, 4).setValue(JSON.stringify(semestersData)); sheet.getRange(rowIndex, 5).setValue(now); } else { // 新增全新學生列 sheet.appendRow([studentId, studentName, grade, JSON.stringify(semestersData), now]); } return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "資料已成功同步儲存至 Google 雲端試算表！" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
